cmake_minimum_required(VERSION 3.22)
project(heming_audio LANGUAGES CXX)
add_library(heming_audio SHARED src/heming_audio.cpp src/preview_player.cpp)
target_compile_features(heming_audio PRIVATE cxx_std_17)
target_include_directories(heming_audio PRIVATE src)
if(WIN32)
  target_compile_options(heming_audio PRIVATE /utf-8 /EHsc)
  target_link_libraries(heming_audio PRIVATE ws2_32 ole32 avrt)
elseif(ANDROID)
  target_link_libraries(heming_audio PRIVATE dl m log atomic aaudio)
endif()
option(HEMING_BUILD_TESTS "Build deterministic native synchronization tests" OFF)
if(HEMING_BUILD_TESTS)
  enable_testing()
  target_compile_definitions(heming_audio PRIVATE HEMING_PREVIEW_TESTING HEMING_TRANSPORT_TESTING)
  add_executable(heming_pcm_transport_test src/tests/pcm_transport_test.cpp)
  target_compile_features(heming_pcm_transport_test PRIVATE cxx_std_17)
  target_include_directories(heming_pcm_transport_test PRIVATE src)
  target_link_libraries(heming_pcm_transport_test PRIVATE heming_audio)
  if(WIN32)
    target_link_libraries(heming_pcm_transport_test PRIVATE ws2_32)
  endif()
  if(MSVC)
    target_compile_options(heming_pcm_transport_test PRIVATE /UNDEBUG /utf-8)
  else()
    target_compile_options(heming_pcm_transport_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME pcm_transport_framing_and_latency COMMAND heming_pcm_transport_test)
  set_tests_properties(pcm_transport_framing_and_latency PROPERTIES TIMEOUT 45)
  add_executable(heming_preview_player_test src/tests/preview_player_test.cpp)
  target_compile_features(heming_preview_player_test PRIVATE cxx_std_17)
  target_include_directories(heming_preview_player_test PRIVATE src)
  target_link_libraries(heming_preview_player_test PRIVATE heming_audio)
  if(MSVC)
    target_compile_options(heming_preview_player_test PRIVATE /UNDEBUG /utf-8)
  else()
    target_compile_options(heming_preview_player_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME local_preview_streaming COMMAND heming_preview_player_test)
  add_executable(heming_hardware_clock_test src/tests/hardware_clock_test.cpp)
  target_compile_features(heming_hardware_clock_test PRIVATE cxx_std_17)
  target_include_directories(heming_hardware_clock_test PRIVATE src)
  if(MSVC)
    target_compile_options(heming_hardware_clock_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(heming_hardware_clock_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME hardware_clock_rate COMMAND heming_hardware_clock_test)
  add_executable(heming_output_timing_test src/tests/output_timing_test.cpp)
  target_compile_features(heming_output_timing_test PRIVATE cxx_std_17)
  target_include_directories(heming_output_timing_test PRIVATE src)
  if(MSVC)
    target_compile_options(heming_output_timing_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(heming_output_timing_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME output_presentation_and_drain COMMAND heming_output_timing_test)
endif()
